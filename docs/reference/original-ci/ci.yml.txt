name: Validate
on:
  push:
    branches: [main]
  pull_request:
permissions:
  contents: read
jobs:
  offline-checks:
    runs-on: ubuntu-latest
    env:
      AI_ALLOW_PAID_CALLS: "false"
      NEXT_TELEMETRY_DISABLED: "1"
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: "24"
          cache: npm
      - run: npm ci --no-audit --no-fund
      - run: npm test
      - run: npm run test:benchmark
      - run: npm run build
      - run: npm run typecheck
